- [ ] **Step 2: Add `learningJourneys` to `/build/concepts` payload**

In `srv/lib/published-concepts-query.js`, the existing helper returns concepts with `teaches`/`requires`/`requiredBy`/`relatedTo`. Add a sibling array `learningJourneys`:

```javascript
// At the relevant point in the existing helper, after the existing arrays are computed:

// Phase 4.1 (#447 §2.6): include learning journeys covering each concept.
const { LearningJourneys, LearningJourneyConceptLinks } = cds.entities('com.sap.developers.ims.external');

// Pull journey-links for all concept IDs in one query
const conceptIds = concepts.map(c => c.ID);
const journeyLinks = await SELECT.from(LearningJourneyConceptLinks)
  .columns('concept_ID', 'journey.slug as journeySlug', 'journey.title as title', 'journey.url as url', 'journey.level as level', 'journey.durationHours as durationHours')
  .where({ concept_ID: { in: conceptIds } });

// Group by concept_ID
const journeysByConceptId = new Map();
for (const link of journeyLinks) {
  if (!journeysByConceptId.has(link.concept_ID)) journeysByConceptId.set(link.concept_ID, []);
  journeysByConceptId.get(link.concept_ID).push({
    slug: link.journeySlug,
    title: link.title,
    url: link.url,
    level: link.level,
    durationHours: link.durationHours,
  });
}

// Attach to each concept in the existing result shape
for (const concept of concepts) {
  concept.learningJourneys = journeysByConceptId.get(concept.ID) ?? [];
}
```

**Important — CSP-safe telemetry**: The project's CSP forbids inline `onclick` handlers. **Do NOT use `onclick="..."`**. Instead, attach a delegated click listener via a separate Hugo-emitted script tag at the bottom of the concept page, OR add it to an existing concept-page Vue island if one exists. The pattern used elsewhere on the site is a single delegated listener at page load:

```html
<script>
  document.addEventListener('DOMContentLoaded', function() {
    var section = document.querySelector('[data-kg-section="learning-journeys"]');
    if (!section) return;
    var conceptSlug = section.getAttribute('data-concept-slug');
    section.querySelectorAll('a[data-journey-slug]').forEach(function(a) {
      a.addEventListener('click', function() {
        try {
          window.dispatchEvent(new CustomEvent('kg.learning_journey.linked_from_concept_page', {
            detail: { conceptSlug: conceptSlug, journeySlug: a.getAttribute('data-journey-slug') }
          }));
        } catch (e) {}
      });
    });
  });
</script>
```

(The script is rendered server-side; no inline-event-handler attribute, so CSP `script-src 'self'` allows it. Verify against the existing concept page's CSP posture by checking `approuter/xs-app.json` or the equivalent if you're unsure.)

```vue
<!-- Phase 4.1 (#447): cross-corpus 'Other resources' section -->
<section v-if="otherResources.length > 0" class="kg-section kg-section-other">
  <h3 class="kg-section-title">Other resources</h3>
  <ul class="kg-list">
    <li v-for="r in otherResources" :key="r.slug" class="kg-item">
      <a :href="r.url" target="_blank" rel="noopener"
         @click="onOtherResourceClick(r)">
        {{ r.title }}
      </a>
      <span v-if="r.level || r.durationHours" class="kg-meta">
        <template v-if="r.level"> · {{ formatLevel(r.level) }}</template>
        <template v-if="r.durationHours"> · {{ r.durationHours }}h</template>
      </span>
    </li>
  </ul>
</section>
```

And in `<script setup>`:

```typescript
import type { OtherResource } from './types';

const otherResources = computed<OtherResource[]>(
  () => props.neighborhood?.otherResources ?? []
);

function formatLevel(level: string): string {
  if (!level) return '';
  return level.charAt(0).toUpperCase() + level.slice(1).toLowerCase();
}

function onOtherResourceClick(r: OtherResource) {
  if (typeof window === 'undefined') return;
  if (r.type === 'learning-journey') {
    window.dispatchEvent(new CustomEvent('kg.learning_journey.linked_from_sidebar', {
      detail: {
        tutorialSlug: props.neighborhood?.tutorial?.slug,
        journeySlug: r.slug,
      },
    }));
  }
  // Future sub-phases: branch on r.type for other telemetry events.
}
```
