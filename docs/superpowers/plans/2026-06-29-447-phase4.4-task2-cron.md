## 2.3 Projection helper + `/build/concepts` payload

- [ ] **Step 13: Extend `srv/lib/kg-projection.js`** — add `buildVideoTriples` mirroring `buildDiscoveryMissionTriples`:

```javascript
export function buildVideoTriples({ videos = [], links = [] } = {}) {
  const triples = [];
  const linksByVideoSlug = new Map();
  for (const l of links) {
    if (!l.videoSlug || !l.conceptSlug) continue;
    if (!linksByVideoSlug.has(l.videoSlug)) linksByVideoSlug.set(l.videoSlug, []);
    linksByVideoSlug.get(l.videoSlug).push(l);
  }
  for (const v of videos) {
    if (!isWithinTTL('video', v.lastSeenAt)) continue;
    const iri = iriVideo(v.slug);
    triples.push([iri, 'rdf:type', 'kg:Video']);
    triples.push([iri, 'kg:title', JSON.stringify(v.title)]);
    triples.push([iri, 'kg:slug', JSON.stringify(v.slug)]);
    if (v.publishedAt) triples.push([iri, 'kg:publishedAt', JSON.stringify(v.publishedAt)]);
    if (v.channelTitle) triples.push([iri, 'kg:channelTitle', JSON.stringify(v.channelTitle)]);

    for (const l of linksByVideoSlug.get(v.slug) ?? []) {
      triples.push([iri, 'kg:teaches', iriConcept(l.conceptSlug)]);
    }
  }
  return triples;
}
```

```javascript
const { Videos, VideoConceptLinks } = cds.entities('com.sap.developers.ims.external');

if (conceptIds.length === 0) {
  for (const c of concepts) c.videos = [];
} else {
  const videoLinks = await SELECT.from(VideoConceptLinks)
    .columns(
      'concept_ID',
      'video.slug as videoSlug',
      'video.title as title',
      'video.url as url',
      'video.thumbnailUrl as thumbnailUrl',
      'video.channelTitle as channelTitle',
      'video.publishedAt as publishedAt',
    )
    .where({ concept_ID: { in: conceptIds } })
    .orderBy('video.publishedAt desc')
    .limit(8 * conceptIds.length);

  const videosByConceptId = new Map();
  for (const link of videoLinks) {
    if (!videosByConceptId.has(link.concept_ID)) videosByConceptId.set(link.concept_ID, []);
    const arr = videosByConceptId.get(link.concept_ID);
    if (arr.length < 8) {
      arr.push({
        slug: link.videoSlug,
        title: link.title,
        url: link.url,
        thumbnailUrl: link.thumbnailUrl,
        channelTitle: link.channelTitle,
        publishedAt: link.publishedAt,
      });
    }
  }
  for (const concept of concepts) {
    concept.videos = videosByConceptId.get(concept.ID) ?? [];
  }
}
```
