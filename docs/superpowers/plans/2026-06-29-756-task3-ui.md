```javascript
// app/admin-shell/webapp/controller/job-controls-helpers.js
//
// Pure-function helpers extracted from Board.controller.js for testability.
// Controller methods stay thin wrappers around these.

/**
 * JOIN listJobs output with JobLastRun rows by jobName.
 * Both inputs are plain arrays of objects; output is a flat array suitable
 * for setting on a JSONModel at '/jobs'.
 *
 * @param {Array<{jobName, schedule, ttlMs, description, nextRunIso}>} jobs
 * @param {Array<{jobName, lastSuccessAt?, lastErrorAt?, lastErrorMessage?}>} lastRuns
 * @returns {Array<object>}
 */
export function joinJobsWithLastRuns(jobs, lastRuns) {
  const byName = new Map((lastRuns || []).map(r => [r.jobName, r]));
  return (jobs || []).map(j => ({
    ...j,
    lastSuccessAt: byName.get(j.jobName)?.lastSuccessAt ?? null,
    lastErrorAt: byName.get(j.jobName)?.lastErrorAt ?? null,
    lastErrorMessage: byName.get(j.jobName)?.lastErrorMessage ?? null,
    isRunning: false,
  }));
}

/**
 * Humanize an ISO timestamp into "in N hours" (< 24h away) OR
 * "Sun 03:13 UTC" (>= 24h away) OR "—" (null/invalid).
 */
export function formatNextRun(iso) {
  if (!iso) return '—';
  const date = new Date(iso);
  if (Number.isNaN(date.getTime())) return '—';
  const diffMs = date.getTime() - Date.now();
  if (diffMs < 0) return 'overdue';
  const diffHrs = Math.round(diffMs / 3600000);
  if (diffHrs < 24) {
    return diffHrs === 0 ? 'in <1 hour' : `in ${diffHrs} hour${diffHrs === 1 ? '' : 's'}`;
  }
  const days = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
  const pad = n => String(n).padStart(2, '0');
  return `${days[date.getUTCDay()]} ${pad(date.getUTCHours())}:${pad(date.getUTCMinutes())} UTC`;
}

/**
 * Humanize an ISO timestamp into "N minutes ago" / "N hours ago" /
 * "N days ago" / "Never" (null/invalid).
 */
export function formatRelativeTime(iso) {
  if (!iso) return 'Never';
  const date = new Date(iso);
  if (Number.isNaN(date.getTime())) return 'Never';
  const diffMs = Date.now() - date.getTime();
  if (diffMs < 0) return 'just now';
  const mins = Math.round(diffMs / 60000);
  if (mins < 1) return 'just now';
  if (mins < 60) return `${mins} minute${mins === 1 ? '' : 's'} ago`;
  const hrs = Math.round(diffMs / 3600000);
  if (hrs < 24) return `${hrs} hour${hrs === 1 ? '' : 's'} ago`;
  const days = Math.round(diffMs / 86400000);
  return `${days} day${days === 1 ? '' : 's'} ago`;
}
```

## 3.4 Wire the controller

- [ ] **Step 9: Extend `Board.controller.js`.** Add:

```javascript
// At top of file, add imports.
// (Adjust the import path style to match the file's existing convention —
// most UI5 controllers use sap.ui.define rather than ES imports. If so,
// register job-controls-helpers.js as a sap.ui.define dependency.)
sap.ui.define([
  // ... existing dependencies ...
  'sap/m/MessageToast',
  'sap/m/MessageBox',
  'sap/ui/model/json/JSONModel',
  'tutorialsIms/adminshell/controller/job-controls-helpers',
], function (
  // ... existing args ...
  MessageToast,
  MessageBox,
  JSONModel,
  helpers
) {
  'use strict';

  return Controller.extend('...', {
    // ... existing methods ...

    /**
     * #756: load + JOIN the JobControls data for the Cron health tile.
     * Called from onInit() or onAfterRendering() as part of the existing
     * tile-load flow.
     */
    async _loadJobControls() {
      const adminModel = this.getOwnerComponent().getModel('admin');
      if (!this.getView().getModel('jobControls')) {
        this.getView().setModel(new JSONModel({ jobs: [] }), 'jobControls');
      }
      const jobControlsModel = this.getView().getModel('jobControls');

      try {
        const [jobsResp, lastRuns] = await Promise.all([
          this._callListJobs(adminModel),
          this._loadJobLastRunRows(adminModel),
        ]);
        const joined = helpers.joinJobsWithLastRuns(jobsResp, lastRuns);
        jobControlsModel.setProperty('/jobs', joined);
      } catch (err) {
        console.warn('_loadJobControls failed:', err.message);
        // Best-effort — the tile renders 0 rows, the rest of the Board still works.
      }
    },

    /**
     * #756: invoke AdminService.JobControls.listJobs(). Returns array
     * of {jobName, schedule, ttlMs, description, nextRunIso}.
     */
    async _callListJobs(adminModel) {
      const action = adminModel.bindContext('/JobControls/AdminService.listJobs(...)');
      await action.execute();
      return action.getBoundContext().getObject().value ?? [];
    },

    /**
     * #756: read all JobLastRun rows (small table — ≤24 rows after pre-seed).
     */
    async _loadJobLastRunRows(adminModel) {
      const binding = adminModel.bindList('/JobLastRun');
      const ctxs = await binding.requestContexts(0, 999);
      return ctxs.map(c => c.getObject());
    },

    /**
     * #756: Run now button press handler. Optimistically sets isRunning,
     * invokes runJob, shows toast, schedules a 5-min poll for completion.
     */
    async onRunJob(oEvent) {
      const ctx = oEvent.getSource().getBindingContext('jobControls');
      const jobName = ctx.getProperty('jobName');
      const idx = parseInt(ctx.getPath().split('/').pop(), 10);
      const model = this.getView().getModel('jobControls');

      // Optimistic UI.
      model.setProperty(`/jobs/${idx}/isRunning`, true);

      try {
        const result = await this._callRunJob(jobName);
        if (result.started) {
          MessageToast.show(`${jobName}: started`);
          this._scheduleJobControlsRefresh();
        } else {
          MessageToast.show(`${jobName}: ${result.reason ?? 'skipped'}`);
          model.setProperty(`/jobs/${idx}/isRunning`, false);
        }
      } catch (err) {
        MessageBox.error(`Failed to start ${jobName}: ${err.message}`);
        model.setProperty(`/jobs/${idx}/isRunning`, false);
      }
    },

    /**
     * #756: invoke AdminService.JobControls.runJob(jobName).
     */
    async _callRunJob(jobName) {
      const adminModel = this.getOwnerComponent().getModel('admin');
      const action = adminModel.bindContext('/JobControls/AdminService.runJob(...)');
      action.setParameter('jobName', jobName);
      await action.execute();
      return action.getBoundContext().getObject();
    },

    /**
     * #756: poll JobLastRun every 30s for the next 5min after a manual
     * trigger. Stops automatically when 5min elapses.
     */
    _scheduleJobControlsRefresh() {
      if (this._jobControlsPollHandle) return;       // already polling
      let elapsed = 0;
      const POLL_INTERVAL_MS = 30000;
      const POLL_MAX_MS = 5 * 60 * 1000;
      this._jobControlsPollHandle = setInterval(async () => {
        elapsed += POLL_INTERVAL_MS;
        try {
          await this._loadJobControls();
        } catch (err) {
          // ignore — best-effort poll
        }
        if (elapsed >= POLL_MAX_MS) {
          clearInterval(this._jobControlsPollHandle);
          this._jobControlsPollHandle = null;
        }
      }, POLL_INTERVAL_MS);
    },

    /**
     * #756: UI formatters bound from the XML view.
     */
    formatNextRun(iso) { return helpers.formatNextRun(iso); },
    formatRelativeTime(iso) { return helpers.formatRelativeTime(iso); },

  });
});
```
