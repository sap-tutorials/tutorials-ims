```html
<script>
  document.addEventListener('DOMContentLoaded', function() {
    /* ... existing handlers (tutorials, learning-journeys, blog-posts) ... */

    // Phase 4.3: discovery-mission link telemetry.
    var missionSection = document.querySelector('[data-kg-section="discovery-missions"]');
    if (missionSection) {
      var conceptSlug = missionSection.getAttribute('data-concept-slug');
      missionSection.querySelectorAll('a[data-mission-slug]').forEach(function(a) {
        a.addEventListener('click', function() {
          try {
            window.dispatchEvent(new CustomEvent('kg.discovery_mission.linked_from_concept_page', {
              detail: { conceptSlug: conceptSlug, missionSlug: a.getAttribute('data-mission-slug') }
            }));
          } catch (e) { /* swallow — telemetry must not block navigation */ }
        });
      });
    }
  });
</script>
```

// @vitest-environment happy-dom
import { describe, it, expect, beforeEach } from 'vitest';
import { mount } from '@vue/test-utils';
import RelatedGraph from '../../../hugo-apps/src/related-graph/RelatedGraph.vue';

// Mirrors the existing Phase 4.2 test pattern: mock globalThis.fetch

function makePayload(otherResources: any[]) {
  return {
    tutorial: { slug: 'cap-handlers-tutorial', title: 'CAP Handlers Tutorial' },
    teaches: [{ slug: 'cap-handlers', name: 'CAP handlers', confidence: 0.9 }],
    prerequisitesOf: [],
    sharedConcepts: [],
    whatToLearnNext: [],
    otherResources,
  };

describe('RelatedGraph "Other resources" — discovery-mission rows', () => {
  beforeEach(() => {
    document.documentElement.dataset.pageSlug = 'cap-handlers-tutorial';
    globalThis.fetch = async () => ({
      ok: true,
      async json() {
        return makePayload([
          {
            type: 'discovery-mission',
            slug: 'dm-3019',
            title: 'Get Started with SAP BTP Enterprise Account',
            url: 'https://discovery-center.cloud.sap/missiondetail/3019/',
            effortLevel: 2,
            categoryLabel: 'Onboarding',
          },
        ]);
      },
    }) as any;
  });

  it('renders discovery-mission rows in Other resources', async () => {
    const wrapper = mount(RelatedGraph);
    await new Promise(r => setTimeout(r, 50));
    expect(wrapper.text()).toContain('Other resources');
    expect(wrapper.text()).toContain('Get Started with SAP BTP Enterprise Account');

  it('renders "· effort N · CategoryLabel" meta for mission rows', async () => {
    const wrapper = mount(RelatedGraph);
    await new Promise(r => setTimeout(r, 50));
    expect(wrapper.text()).toContain('effort 2');
    expect(wrapper.text()).toContain('Onboarding');
  });

  it('does NOT render learning-journey or blog-post meta for a mission row', async () => {
    const wrapper = mount(RelatedGraph);
    await new Promise(r => setTimeout(r, 50));
    // mutual-exclusion: mission rows should not surface level/duration/author/date
    expect(wrapper.text()).not.toMatch(/Beginner|Intermediate|Advanced/);
    expect(wrapper.text()).not.toMatch(/\dh\b/);   // no "Nh" duration
    expect(wrapper.text()).not.toMatch(/by .+ ·/);  // no "by Author ·"

  it('renders external link with target=_blank rel=noopener', async () => {
    const wrapper = mount(RelatedGraph);
    await new Promise(r => setTimeout(r, 50));
    const link = wrapper.find('a[href="https://discovery-center.cloud.sap/missiondetail/3019/"]');

    expect(link.attributes('target')).toBe('_blank');
    expect(link.attributes('rel')).toContain('noopener');

  it('fires kg.discovery_mission.linked_from_sidebar on click', async () => {

    const wrapper = mount(RelatedGraph);
    await new Promise(r => setTimeout(r, 50));
    await wrapper.find('a[href="https://discovery-center.cloud.sap/missiondetail/3019/"]').trigger('click');

    expect(events.length).toBeGreaterThan(0);
    expect(events[0]).toMatchObject({
      tutorialSlug: 'cap-handlers-tutorial',
      missionSlug: 'dm-3019',
    });

    window.removeEventListener('kg.discovery_mission.linked_from_sidebar', listener as EventListener);
  });
});
