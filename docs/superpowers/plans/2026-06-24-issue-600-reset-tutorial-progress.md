# Reset Tutorial Progress — Implementation Plan (Issue #600)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a learner reset their progress on a single tutorial (with confirmation), preserve the previous completion in their `/me/` history, and let them complete the tutorial fresh.

**Architecture:** Add `attemptNumber` column + `SUPERSEDED` status enum value to `TaskRecords`. A new CAP action `resetTutorialProgress(slug)` marks the user's live attempt SUPERSEDED (preserving `completionDate`) and inserts a fresh TUTORIAL-level row at attempt N+1. Every TaskRecord read site is updated to handle the two semantics ("current-attempt" vs "has-ever-completed", deduped where appropriate). A small Vue island on the tutorial page surfaces a confirmation dialog and triggers a full page reload after server-side reset.

**Tech Stack:** CAP Node.js, HANA + HDI migration tables, CDS QL, Vue 3 + UI5 web components, Vitest.

**Spec:** [docs/superpowers/specs/2026-06-24-issue-600-reset-tutorial-progress-design.md](../specs/2026-06-24-issue-600-reset-tutorial-progress-design.md)

**Closes:** [#600](https://github.com/sap-tutorials/tutorials-ims/issues/600)

```javascript
this.on('resetTutorialProgress', async (req) => {
  const { slug } = req.data;
  const user = req.user || cds.context?.user;
  if (!user) return req.reject(401, 'Unauthenticated');

  const sapId = resolveUserSapId(user);
  if (!sapId) return req.reject(401, 'Unauthenticated');

  const db = await cds.connect.to('db');
  const { Users: dbUsers, Tutorials: dbTutorials, Steps: dbSteps, TaskRecords: dbTaskRecords } =
    cds.entities('com.sap.developers.ims');

  // 1. Resolve slug → tutorial
  const tutorial = await SELECT.one.from(dbTutorials).where({ slug });
  if (!tutorial) return req.reject(404, `Tutorial not found: ${slug}`);

  // 2. Resolve user (sapId → dbUser)
  const dbUser = await SELECT.one.from(dbUsers).where({ sapId });
  if (!dbUser) {
    // First-time user, nothing to reset.
    return { newAttemptNumber: 1, previousAttemptCompletedAt: null, supersededRecordCount: 0 };
  }

  // 3. Find live (non-SUPERSEDED) rows for this tutorial's steps + tutorial-level
  const steps = await SELECT.from(dbSteps).where({ tutorial_ID: tutorial.ID });
  const taskLegacyIds = [...steps.map(s => s.legacyId), tutorial.legacyId];

  const liveRows = await SELECT.from(dbTaskRecords).where({
    user_ID: dbUser.ID,
    taskLegacyId: { in: taskLegacyIds },
    status: { '!=': 'SUPERSEDED' },
  });

  if (liveRows.length === 0) {
    return { newAttemptNumber: 1, previousAttemptCompletedAt: null, supersededRecordCount: 0 };
  }

  // 4. Determine next attempt number
  const maxAttempt = Math.max(...liveRows.map(r => r.attemptNumber ?? 1));

  // Capture the prior tutorial-level completion date BEFORE we update.
  const priorTutorialRow = liveRows.find(
    r => r.taskType === 'TUTORIAL' && r.status === 'COMPLETED'
  );
  const previousAttemptCompletedAt = priorTutorialRow?.completionDate ?? null;

  // 5. Pre-allocate legacyId for the new row (fail-fast on sequence issues)
  const newLegacyId = await getNextLegacyId('TaskRecords', db);

  // 6. Supersede current rows in one UPDATE
  await UPDATE(dbTaskRecords)
    .set({ status: 'SUPERSEDED' })
    .where({ ID: { in: liveRows.map(r => r.ID) } });

  // 7. Insert fresh TUTORIAL-level row at attempt+1
  await INSERT.into(dbTaskRecords).entries({
    user_ID: dbUser.ID,
    taskLegacyId: tutorial.legacyId,
    taskType: 'TUTORIAL',
    status: 'IN_PROGRESS',
    progress: 0,
    attemptNumber: maxAttempt + 1,
    titleSnapshot: tutorial.title,
    legacyId: newLegacyId,
  });

  // 8. Emit audit event for traceability (the @PersonalData annotation
  //    already audits the writes themselves; this surfaces INTENT).
  await cds.emit('TutorialProgressReset', {
    user: dbUser.ID,
    tutorialSlug: slug,
    attemptNumber: maxAttempt + 1,
    supersededRecordCount: liveRows.length,
    previousAttemptCompletedAt,
  });

  return {
    newAttemptNumber: maxAttempt + 1,
    previousAttemptCompletedAt,
    supersededRecordCount: liveRows.length,
  };
});
```

```vue
<script setup lang="ts">
import { ref, onMounted } from 'vue';

interface Props { slug: string }
const props = defineProps<Props>();

const showReset = ref(false);
const dialogOpen = ref(false);
const submitting = ref(false);
const errorMessage = ref('');

onMounted(async () => {
  try {
    const res = await fetch(`/api/getProgress?slug=${encodeURIComponent(props.slug)}`, {
      credentials: 'include',
    });
    if (!res.ok) return;
    const data = await res.json();
    // Tutorial considered complete when all steps are listed in completedSteps
    // AND the total step count matches. Server-side `getProgress` only returns
    // current-attempt completions (Task 6), so a reset already returned to
    // an empty completedSteps state.
    const stepCount = Number(document.documentElement.dataset.stepCount ?? '0');
    if (stepCount > 0 && data.completedSteps?.length === stepCount) {
      showReset.value = true;
    }
  } catch {
    /* silent — if /api/getProgress fails, the reset button just stays hidden */
  }
});

function openConfirm() { dialogOpen.value = true; errorMessage.value = ''; }
function closeConfirm() { dialogOpen.value = false; }

async function confirmReset() {
  if (submitting.value) return;
  submitting.value = true;
  errorMessage.value = '';
  try {
    const res = await fetch('/api/resetTutorialProgress', {
      method: 'POST',
      credentials: 'include',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ slug: props.slug }),
    });
    if (res.status === 429) {
      errorMessage.value = "You've reset progress too many times — please wait a few minutes.";
      return;
    }
    if (!res.ok) {
      const text = await res.text().catch(() => '');
      errorMessage.value = text || 'Reset failed. Please try again.';
      return;
    }
    // Dispatch + reload in the SAME synchronous block.
    document.dispatchEvent(new CustomEvent('tutorial-reset', { detail: { slug: props.slug } }));
    window.location.reload();
  } catch (err) {
    errorMessage.value = err instanceof Error ? err.message : 'Network error';
  } finally {
    submitting.value = false;
  }
}
</script>

<template>
  <div v-if="showReset" class="tutorial-reset-widget">
    <ui5-button design="Default" icon="restart" @click="openConfirm">
      Reset progress and try again
    </ui5-button>
    <ui5-dialog
      :open="dialogOpen"
      header-text="Reset progress?"
      @close="closeConfirm"
    >
      <p>
        Resetting will clear your progress on this tutorial so you can complete
        it again. Your previous completion stays in your past-completion history.
      </p>
      <ui5-message-strip
        v-if="errorMessage"
        design="Negative"
        hide-close-button
      >
        {{ errorMessage }}
      </ui5-message-strip>
      <template #footer>
        <ui5-button @click="closeConfirm">Cancel</ui5-button>
        <ui5-button
          design="Negative"
          @click="confirmReset"
          :disabled="submitting"
        >
          Reset progress
        </ui5-button>
      </template>
    </ui5-dialog>
  </div>
</template>
```

```html
<script>
document.addEventListener('tutorial-reset', function (e) {
  var slug = e.detail && e.detail.slug;
  if (!slug) return;
  var prefix = 'tutorial-validation-' + slug + '-';
  var toRemove = [];
  for (var i = 0; i < localStorage.length; i++) {
    var k = localStorage.key(i);
    if (k && k.indexOf(prefix) === 0) toRemove.push(k);
  }
  toRemove.forEach(function (k) { localStorage.removeItem(k); });
  // Clear data-validated attributes on rendered step containers
  document.querySelectorAll('.tutorial-step[data-validated="true"]').forEach(function (el) {
    el.removeAttribute('data-validated');
  });
});
</script>
```
