{{- $src := .Destination -}} {{- $alt := .PlainText -}} {{- $title := .Title -}} {{- $attrs := .Attributes | default dict -}} {{- $width := index $attrs "width" | default "" -}} {{- $height := index $attrs "height" | default "" -}} {{- $caption := index $attrs "caption" | default "" -}} {{- $hasCaption := or (and (ne $alt "") (ne $alt "image")) (ne $caption "") -}} {{- $captionText := $caption -}} {{- if eq $captionText "" -}}{{- $captionText = $alt -}}{{- end -}} {{- /* Look up intrinsic dimensions from sidecar (populated by fetch-tutorials). Markdown attribute syntax can't be used for inline images — goldmark emits `{width="..."}` as literal text — so we resolve dims by URL here. */ -}} {{- if and (eq $width "") (eq $height "") -}} {{- $dims := index site.Data.image_dimensions $src -}} {{- if $dims -}} {{- $width = printf "%v" $dims.w -}} {{- $height = printf "%v" $dims.h -}} {{- end -}} {{- end -}} {{- $useCdn := and (hasPrefix $src "https://raw.githubusercontent.com/") (eq (printf "%s" $src) $src) -}} {{- $finalSrc := $src -}} {{- $srcset := "" -}} {{- if $useCdn -}} {{- $enc := $src | urlquery -}} {{- $finalSrc = printf "/img-cdn/?u=%s&w=1440" $enc -}} {{- $srcset = printf "/img-cdn/?u=%s&w=480 480w, /img-cdn/?u=%s&w=960 960w, /img-cdn/?u=%s&w=1440 1440w" $enc $enc $enc -}} {{- end -}} {{- if $hasCaption -}}{{- end -}}
